CREATE TABLE Category (
    category_id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL
);

CREATE TABLE Location (
    location_id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    lats VARCHAR(50),
    langs VARCHAR(50),
    userId INT NOT NULL
);

CREATE TABLE Status (
    status_id SERIAL PRIMARY KEY,
    message VARCHAR(50) NOT NULL,
    description TEXT
);

CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    timestamp TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    name VARCHAR(100) NOT NULL,
    username VARCHAR(100) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    phone VARCHAR(50),
    password VARCHAR(255) NOT NULL,
    city VARCHAR(100),
    photo_id BIGINT,
    user_role VARCHAR(50) NOT NULL
);

CREATE TABLE tag (
    tag_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    tag_type VARCHAR(50)
);

CREATE TABLE skill (
    skill_id SERIAL PRIMARY KEY,
    category_id BIGINT,
    title VARCHAR(255) NOT NULL,
    sub_title VARCHAR(255),
    faqs TEXT,
    long_description TEXT,
    photo_ids TEXT,
    logo_photo_id VARCHAR(50),

    CONSTRAINT fk_category FOREIGN KEY (category_id)
    REFERENCES category (category_id)
);

CREATE TABLE review (
    review_id SERIAL PRIMARY KEY,
    rating INT NOT NULL,
    appointment_id INT,
    location_id INT,
    status_id INT,

    CONSTRAINT fk_appointment
    FOREIGN KEY(appointment_id)
    REFERENCES appointment(appointment_id),

    CONSTRAINT fk_location
    FOREIGN KEY(location_id)
    REFERENCES location(location_id),

    CONSTRAINT fk_status
    FOREIGN KEY(status_id)
    REFERENCES status(status_id)
);

CREATE TABLE Worker (
    worker_id VARCHAR(50) PRIMARY KEY,
    timestamp TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    user_id INT NOT NULL,
    bio TEXT,
    title VARCHAR(100),
    experience TEXT,
    total_rating DOUBLE PRECISION,
    total_review_count INTEGER,
    photo_id VARCHAR(50),
    CONSTRAINT fk_user FOREIGN KEY (user_id) REFERENCES Users(user_id)
);

CREATE TABLE faq (
    id SERIAL PRIMARY KEY,
    question VARCHAR(255) NOT NULL,
    category_id BIGINT NOT NULL,
    skill_id BIGINT,
    answer TEXT NOT NULL,
    CONSTRAINT fk_category FOREIGN KEY (category_id) REFERENCES category (category_id),
    CONSTRAINT fk_skill FOREIGN KEY (skill_id) REFERENCES skill (skill_id)
);

-- Create join tables for the many-to-many relationships
CREATE TABLE WorkerSkill (
    worker_id VARCHAR(50),
    skill_id INT,
    PRIMARY KEY (worker_id, skill_id),
    CONSTRAINT fk_worker FOREIGN KEY (worker_id) REFERENCES Worker(worker_id),
    CONSTRAINT fk_skill FOREIGN KEY (skill_id) REFERENCES Skill(skill_id)
);

-- Create join tables for the many-to-many relationships
CREATE TABLE WorkerReview (
    worker_id VARCHAR(50),
    review_id INT,
    PRIMARY KEY (worker_id, review_id),
    CONSTRAINT fk_worker FOREIGN KEY (worker_id) REFERENCES Worker(worker_id),
    CONSTRAINT fk_review FOREIGN KEY (review_id) REFERENCES Review(review_id)
);

-- Create join tables for the many-to-many relationships
CREATE TABLE WorkerTag (
    worker_id VARCHAR(50),
    tag_id int,
    PRIMARY KEY (worker_id, tag_id),
    CONSTRAINT fk_worker FOREIGN KEY (worker_id) REFERENCES Worker(worker_id),
    CONSTRAINT fk_tag FOREIGN KEY (tag_id) REFERENCES Tag(tag_id)
);

CREATE TABLE UserTag (
    user_id INT,
    tag_id INT,
    PRIMARY KEY (user_id, tag_id),
    CONSTRAINT fk_user FOREIGN KEY (user_id) REFERENCES Users(user_id),
    CONSTRAINT fk_tag FOREIGN KEY (tag_id) REFERENCES tag(tag_id)
);
